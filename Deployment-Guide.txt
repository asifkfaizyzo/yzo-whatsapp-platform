# ============================================================
# DOCKER COMMANDS CHEAT SHEET
# ============================================================
# SudoReply runs on Docker, not PM2. These replace all the
# pm2 commands you're used to from Cureli.
#
# All commands run from: /var/www/SudoReply
# ----------------------------------------------------------
#
# STATUS & INFO
#   docker compose ps                          # all containers (like pm2 list)
#   docker stats                               # CPU/RAM per container (like pm2 monit)
#   docker compose inspect backend             # full details (IP, env, mounts)
#
# LOGS
#   docker compose logs -f                     # all containers (like pm2 logs)
#   docker compose logs -f backend             # backend only
#   docker compose logs -f admin-web           # admin frontend
#   docker compose logs -f tenant-web          # tenant frontend
#   docker compose logs -f postgres            # database
#   docker compose logs -f redis               # cache
#   docker compose logs --tail 100 backend     # last 100 lines, no tailing
#   docker compose logs --since 1h backend     # last 1 hour
#   docker compose logs backend 2>&1 | grep -i error   # search for errors
#
# START / STOP / RESTART
#   docker compose up -d                       # start everything
#   docker compose down                        # stop everything (keeps data)
#   docker compose down -v                     # STOP + DELETE DATABASE (danger!)
#   docker compose restart backend             # restart one service
#   docker compose restart                     # restart everything
#   docker compose stop backend                # stop one service
#
# REBUILD & DEPLOY
#   docker compose up -d --build backend       # rebuild + restart backend
#   docker compose up -d --build               # rebuild + restart all
#   docker compose build --no-cache backend    # rebuild without cache
#
# RUN COMMANDS INSIDE CONTAINERS
#   docker compose exec backend sh             # shell into backend
#   docker compose exec backend npx prisma migrate deploy   # run migrations
#   docker compose exec backend npx prisma generate         # regen client
#   docker compose exec backend node scripts/something.js   # run a script
#   docker compose exec postgres psql -U yzoadmin -d fullweb  # DB shell
#
# DATABASE
#   docker compose exec postgres psql -U yzoadmin -d fullweb -c "SELECT count(*) FROM users;"
#   docker compose exec postgres pg_dump -U yzoadmin fullweb > backup.sql    # backup
#   cat backup.sql | docker compose exec -T postgres psql -U yzoadmin fullweb  # restore
#
# CLEANUP
#   docker system df                           # disk usage
#   docker system prune                        # clean old images/containers
#
# PM2 → DOCKER TRANSLATION
#   pm2 list              →  docker compose ps
#   pm2 logs              →  docker compose logs -f
#   pm2 logs --lines 50   →  docker compose logs --tail 50 backend
#   pm2 restart app       →  docker compose restart backend
#   pm2 stop app          →  docker compose stop backend
#   pm2 start app         →  docker compose up -d backend
#   pm2 delete app        →  docker compose down
#   pm2 monit             →  docker stats
# ============================================================


# ============================================================
# NOTES
# ============================================================

# .env files are NOT in git — they live only on the VPS.
# Never overwrite them during deploy.
# Locations:
#   /var/www/SudoReply/.env          (root — VITE_* vars for frontends)
#   /var/www/SudoReply/backend/.env  (backend — DB, Redis, secrets)
#
# If you add new VITE_* env vars locally, you must:
#   1. Add them to /var/www/SudoReply/.env on the VPS
#   2. Rebuild the frontend Docker images (Step 6)
#   VITE_* vars are baked at BUILD time, not runtime.
#
# If you add new backend env vars, you must:
#   1. Add them to /var/www/SudoReply/backend/.env on the VPS
#   2. Restart the backend container (Step 5)
#
# Prisma:
#   "npx prisma migrate deploy" runs pending migrations only.
#   It does NOT reset data. Safe to run on every deploy.
#   Run it INSIDE the container: docker compose exec backend ...
#
# Docker Compose vs PM2:
#   Unlike Cureli (which uses PM2 + static files), SudoReply
#   runs everything in Docker containers on the VPS.
#   Frontends are served by nginx INSIDE their containers,
#   not from /var/www/ static directories.
#   The host-level nginx at /etc/nginx/sites-enabled/sudoreply.conf
#   just reverse-proxies to the container ports.
#
# IMPORTANT — "docker compose" vs "docker-compose":
#   Use "docker compose" (SPACE, v2 plugin) — this is what's
#   installed on the VPS and it works correctly.
#   Do NOT use "docker-compose" (HYPHEN, v1 standalone) — it's
#   an old version (1.29.2) with a known ContainerConfig bug
#   that will break container recreation.
#
# Database:
#   PostgreSQL 16 runs INSIDE Docker (port 5432 internally).
#   Connection string uses internal Docker hostname:
#     DATABASE_URL=postgresql://yzoadmin:Sibilissy@5453@postgres:5432/fullweb
#   To access DB directly:
#     docker compose exec postgres psql -U yzoadmin -d fullweb
#   NOTE: Port 5432 on the host is SudoReply's Docker PG.
#         Port 5433 on the host is YZO's native PG. Don't mix them.
#
# Redis:
#   Also inside Docker. Hostname: redis, Port: 6379


# ============================================================
# RECOVERY — Accidentally coded on prod instead of dev
# ============================================================
#
# Scenario:
# You've been working for hours on the "prod" branch with tons
# of uncommitted changes. Move ALL of it safely to "dev".
#
# 1. Check current branch:
git branch --show-current

# 2. Stash EVERYTHING (the -u flag is critical — it includes
#    new/untracked files like new components, routes, etc.):
git stash push -u -m "WIP: moving work from prod to dev"

# 3. Verify prod is clean:
git status
# should say: nothing to commit, working tree clean

# 4. Switch to dev:
git switch dev
git pull origin dev

# 5. Restore your changes:
git stash pop

# 6. Verify:
git status

# 7. Commit and push:
git add .
git commit -m "Your commit message"
git push origin dev
#
# Troubleshooting:
#   "git stash pop" says CONFLICT:
#     Open conflicted files, resolve <<<<<<< markers, then:
#       git add . && git stash drop && git commit -m "resolved"
#
#   Stashed WITHOUT -u and lost new files:
#     They're still on prod as untracked files. Go back:
#       git switch prod
#       git stash push -u -m "grabbing missed files"
#       git switch dev
#       git stash pop
#
#   See what's in a stash before popping:
#     git stash list
#     git stash show stash@{0} --stat
# ============================================================

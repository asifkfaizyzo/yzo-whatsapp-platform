# ============================================================
# STEP 5 — Backend: rebuild and restart
# ============================================================
# Only needed if backend code or migrations changed.

cd /var/www/SudoReply

docker compose up -d --build backend

# Run pending migrations (safe — does NOT reset data):
docker compose exec backend npx prisma migrate deploy

# Regenerate Prisma client inside container:
docker compose exec backend npx prisma generate

# Verify:
docker compose logs backend --tail 30


# ============================================================
# STEP 6 — Frontends: rebuild and restart
# ============================================================
# Needed if ANY frontend code changed OR if VITE_* env vars changed.
# The VITE_* vars from root .env are baked in at BUILD time.

cd /var/www/SudoReply

docker compose up -d --build admin-web tenant-web

# Verify containers are running:
docker compose ps

# Verify in browser:
#   https://admin.sudoreply.com
#   https://sudoreply.com


# ============================================================
# STEP 7 — Reload nginx (only if nginx config changed)

# ============================================================
# NOTES
# ============================================================

# .env files are NOT in git — they live only on the VPS.
# Never overwrite them during deploy.
# Locations:
#   /var/www/SudoReply/.env          (root — VITE_* vars for frontends)
#   /var/www/SudoReply/backend/.env  (backend — DB, Redis, secrets)
#
# If you add new VITE_* env vars locally, you must:
#   1. Add them to /var/www/SudoReply/.env on the VPS
#   2. Rebuild the frontend Docker images (Step 6)
#   VITE_* vars are baked at BUILD time, not runtime.
#
# If you add new backend env vars, you must:
#   1. Add them to /var/www/SudoReply/backend/.env on the VPS
#   2. Restart the backend container (Step 5)
#
# Prisma:
#   "npx prisma migrate deploy" runs pending migrations only.
#   It does NOT reset data. Safe to run on every deploy.
#   Run it INSIDE the container: docker compose exec backend ...
#
# Docker Compose vs PM2:
#   Unlike Cureli (which uses PM2 + static files), SudoReply
#   runs everything in Docker containers on the VPS.
#   Frontends are served by nginx INSIDE their containers,
#   not from /var/www/ static directories.
#   The host-level nginx at /etc/nginx/sites-enabled/sudoreply.conf
#   just reverse-proxies to the container ports.
#
# IMPORTANT — "docker compose" vs "docker-compose":
#   Use "docker compose" (SPACE, v2 plugin) — this is what's
#   installed on the VPS and it works correctly.
#   Do NOT use "docker-compose" (HYPHEN, v1 standalone) — it's
#   an old version (1.29.2) with a known ContainerConfig bug
#   that will break container recreation.
#
# Database:
#   PostgreSQL 16 runs INSIDE Docker (port 5432 internally).
#   Connection string uses internal Docker hostname:
#     DATABASE_URL=postgresql://yzoadmin:Sibilissy@5453@postgres:5432/fullweb
#   To access DB directly:
#     docker compose exec postgres psql -U yzoadmin -d fullweb
#   NOTE: Port 5432 on the host is SudoReply's Docker PG.
#         Port 5433 on the host is YZO's native PG. Don't mix them.
#
# Redis:
#   Also inside Docker. Hostname: redis, Port: 6379
#
# Logs:
#   docker compose logs -f backend
#   docker compose logs -f admin-web
#   docker compose logs -f tenant-web
#   docker compose logs -f --tail 50    (all services)
#
# Start/Stop everything:
#   cd /var/www/SudoReply && docker compose up -d
#   cd /var/www/SudoReply && docker compose down

#
# 1. Check current branch:

# 2. Stash changes:

# 3. Switch to dev:
git switch dev

# 4. Restore changes:
git stash pop

# 5. Verify:
git status

# 6. Commit and push:
git add .
git commit -m "Your commit message"
git push origin dev
